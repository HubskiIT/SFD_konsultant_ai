function addToCartFromChat(productKey, selectId) {

const selectElement = document.getElementById(selectId);
const chosenOption = selectElement ?
selectElement.options[selectElement.selectedIndex].text : '';
let price = product.price;
let variantName = chosenOption.split(' - ')[0];
if (chosenOption.includes('+35,00 zł')) {
price = 84.99;
} else if (chosenOption.includes('+25,00 zł')) {
price = 64.99;
}
cart.push({

});
updateCartUI();
showCustomAlert("Dodano do koszyka!", `${product.name}
(${variantName}) pomyślnie dodany do Twojego zamówienia.`, 'success');
triggerCrossSellResponse(productKey);
}

function updateCartUI() {
const cartItemsContainer =
document.getElementById('cart-items');
const cartEmptyState =
document.getElementById('cart-empty-state');
const cartBadge = document.getElementById('cart-badge');
const cartCountBadge =
document.getElementById('cart-count-badge');
const totalHeader =
document.getElementById('cart-total-header');
const subtotalLabel =
document.getElementById('summary-subtotal');
const totalLabel = document.getElementById('summary-total');
if (cart.length === 0) {
cartEmptyState.classList.remove('hidden');
cartBadge.classList.add('opacity-0');
cartCountBadge.innerText = "0 szt.";
totalHeader.innerText = "0,00 zł";
subtotalLabel.innerText = "0,00 zł";
totalLabel.innerText = "0,00 zł";
const itemsList =
cartItemsContainer.querySelectorAll('.cart-item-row');
itemsList.forEach(item => item.remove());
return;
}
cartEmptyState.classList.add('hidden');
cartBadge.classList.remove('opacity-0');
cartBadge.innerText = cart.length;
cartCountBadge.innerText = `${cart.length} szt.`;
let totalSum = 0;
const itemsList =
cartItemsContainer.querySelectorAll('.cart-item-row');
itemsList.forEach(item => item.remove());
cart.forEach(item => {
totalSum += item.price;
const itemHtml = `
<div class="cart-item-row flex items-center gap-3
bg-slate-50 border border-slate-100 rounded-xl p-2.5 transition-all
hover:border-slate-200">
<img src="${item.img}" class="w-10 h-10
rounded-md object-cover shadow-xs" alt="Product">
<div class="flex-1 min-w-0">
<h5 class="font-bold text-xs text-slate-800
truncate">${item.name}</h5>
<p class="text-[10px] text-slate-400
font-medium">${item.variant}</p>
</div>
<div class="text-right shrink-0">
<p class="text-xs font-bold
text-sfdBlue">${item.price.toFixed(2)} zł</p>
<button
onclick="removeFromCart('${item.id}')" class="text-slate-400
hover:text-red-500 text-[10px] underline font-medium
transition-colors">Usuń</button>
</div>
</div>
`;
cartItemsContainer.insertAdjacentHTML('beforeend',
itemHtml);
});
totalHeader.innerText = `${totalSum.toFixed(2)} zł`;
subtotalLabel.innerText = `${totalSum.toFixed(2)} zł`;
totalLabel.innerText = `${totalSum.toFixed(2)} zł`;
}
function removeFromCart(itemId) {

updateCartUI();
}
